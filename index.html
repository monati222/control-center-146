خطوات الربط لـ Google Apps Script & Sheets

  1. أنشئ ملف Google Sheet جديد.
  2. من القائمة اختر Extensions > Apps Script واحذف أي كود افتراضي.
  3. ألصق كود الـ Backend المرفق أدناه في محرر Apps Script.
  4. أضف في Project Settings > Script Properties خاصية باسم TEACHER_EMAIL وضع بريدك الإلكتروني.
  5. انقر فوق Deploy > New Deployment واختر نوع Web App (Execute as: Me, Who has access: Anyone).
  6. انسخ الرابط الناتج (Web App URL) وضعه في متغير BACKEND_URL في الكود أدناه.

كود Google Apps Script المقترح (Backend):

const ss = SpreadsheetApp.getActiveSpreadsheet();
const sheet = ss.getSheetByName("Leaderboard") || ss.insertSheet("Leaderboard");

function doPost(e) {
  try {
    const data = JSON.parse(e.postData.contents);
    const action = data.action;
    
    if (action === "sync") {
      return ContentService.createTextOutput(JSON.stringify(syncRecord(data))).setMimeType(ContentService.MimeType.JSON);
    } else if (action === "leaderboard") {
      return ContentService.createTextOutput(JSON.stringify(getLeaderboardData(data.sessionId))).setMimeType(ContentService.MimeType.JSON);
    } else if (action === "email") {
      return ContentService.createTextOutput(JSON.stringify(sendEmailReport(data.sessionId))).setMimeType(ContentService.MimeType.JSON);
    }
    return ContentService.createTextOutput(JSON.stringify({error: "Invalid action"})).setMimeType(ContentService.MimeType.JSON);
  } catch(err) {
    return ContentService.createTextOutput(JSON.stringify({error: err.toString()})).setMimeType(ContentService.MimeType.JSON);
  }
}

function doGet(e) {
  return ContentService.createTextOutput(JSON.stringify({status: "Active"})).setMimeType(ContentService.MimeType.JSON);
}

function getActiveSessionId() {
  const today = new Date().toISOString().split('T')[0];
  return "SES_" + today;
}

function syncRecord(data) {
  const sessionId = data.sessionId || getActiveSessionId();
  const leaderName = data.leaderName;
  const rows = sheet.getDataRange().getValues();
  
  let foundRow = -1;
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0] === sessionId && rows[i][1] === leaderName) {
      foundRow = i + 1;
      break;
    }
  }
  
  const now = new Date().toISOString();
  const score = Number(data.totalScore) || 0;
  if (foundRow > 0) {
    sheet.getRange(foundRow, 3).setValue(score);
    sheet.getRange(foundRow, 4).setValue(data.status);
    sheet.getRange(foundRow, 5).setValue(data.currentActivity);
    sheet.getRange(foundRow, 7).setValue(now);
    if(data.status === "مكتمل") sheet.getRange(foundRow, 8).setValue(now);
  } else {
    sheet.appendRow([sessionId, leaderName, score, data.status, data.currentActivity, now, now, ""]);
  }
  return {success: true, sessionId: sessionId};
}

function getLeaderboardData(sessionId) {
  const sid = sessionId || getActiveSessionId();
  const rows = sheet.getDataRange().getValues();
  let list = [];
  for (let i = 1; i < rows.length; i++) {
    if (rows[i][0] === sid) {
      list.push({
        sessionId: rows[i][0],
        leaderName: rows[i][1],
        totalScore: Number(rows[i][2]) || 0,
        status: rows[i][3],
        currentActivity: Number(rows[i][4]) || 1,
        startedAt: rows[i][5],
        lastUpdatedAt: rows[i][6],
        finishedAt: rows[i][7]
      });
    }
  }
  list.sort((a,b) => b.totalScore - a.totalScore || (a.finishedAt || "9999").localeCompare(b.finishedAt || "9999"));
  return {success: true, leaderboard: list};
}

function sendEmailReport(sessionId) {
  const email = PropertiesService.getScriptProperties().getProperty("TEACHER_EMAIL");
  if (!email) return {success: false, error: "Teacher email not set"};
  
  const data = getLeaderboardData(sessionId);
  let html = "<h2>تقرير مركز التحكم الذكي - الثانوية 146</h2><table border='1' cellpadding='8' style='border-collapse:collapse;'><tr><th>الترتيب</th><th>القائدة</th><th>النقاط</th><th>الحالة</th></tr>";
  
  data.leaderboard.forEach((item, idx) => {
    html += `<tr><td>${idx+1}</td><td>${item.leaderName}</td><td>${item.totalScore}</td><td>${item.status}</td></tr>`;
  });
  html += "</table>";
  
  GmailApp.sendEmail(email, "تقرير نتائج حصة التقنية الرقمية - مركز التحكم 146", "", {htmlBody: html});
  return {success: true};
}
